"""Archived experiment source; not imported by active benchmarks."""


def admit(tree, enqueue, low, high):
    if not tree.enable_svd_compression:
        return
    pool = tree.req_to_token_pool.mamba_pool
    if not 0 <= low < high <= pool.size:
        raise ValueError('Require 0 <= low < high <= full-state pool size')
    free = pool.available_size()
    # Every live pending snapshot can release one cached full-state slot.
    # Eviction/reset cancellation already removes stale pending jobs.
    projected = free + len(tree._pending_compression)
    active = getattr(tree, '_admission_active', False)
    if projected >= high:
        tree._admission_active = False
        return
    if not active:
        if free > low:
            return
        tree._admission_active = True
        tree._admission_episodes = getattr(tree, '_admission_episodes', 0) + 1
    lru = tree.mamba_lru_list
    node = getattr(lru.tail, lru.prv)
    # Snapshotting does not mutate the LRU list. Commit/eviction stay scheduler-owned.
    while node is not lru.head and projected < high:
        previous = getattr(node, lru.prv)
        if (node.mamba_value is not None and not node.mamba_compressed
                and node.id not in tree._pending_compression
                and tree._is_live_compression_node(node)):
            enqueue(tree, node)
            if node.id in tree._pending_compression:
                projected += 1
                tree._admission_jobs = getattr(tree, '_admission_jobs', 0) + 1
        node = previous
    if projected >= high:
        tree._admission_active = False


def install(policy, low=8, high=12):
    if policy not in ('eager', 'pressure'):
        raise ValueError(f'Unknown compression admission policy: {policy}')
    if policy == 'eager':
        return
    if not 0 <= low < high:
        raise ValueError('Require 0 <= low < high')
    from sglang.srt.mem_cache.mamba_radix_cache import MambaRadixCache
    enqueue = MambaRadixCache._enqueue_compression
    drain = MambaRadixCache.drain_compression_completions
    reset = MambaRadixCache._reset_compression_state

    def pressure_enqueue(tree, node):
        # The new node is already in the full-state LRU. Revisit deferred nodes too.
        admit(tree, enqueue, low, high)

    def pressure_drain(tree, max_per_call=8):
        drain(tree, max_per_call=max_per_call)
        admit(tree, enqueue, low, high)

    def pressure_reset(tree):
        reset(tree)
        tree._admission_active = False
        tree._admission_episodes = 0
        tree._admission_jobs = 0

    MambaRadixCache._enqueue_compression = pressure_enqueue
    MambaRadixCache.drain_compression_completions = pressure_drain
    MambaRadixCache._reset_compression_state = pressure_reset
