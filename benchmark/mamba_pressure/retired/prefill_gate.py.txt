"""Archived strict exclusion experiment; not installed by active servers."""
from contextlib import contextmanager, nullcontext
import threading
import time


class PrefillGate:
    def __init__(self):
        self.condition = threading.Condition()
        self.active = None
        self.prefill_waiters = 0
        self.wait_ms = {'prefill': 0.0, 'svd': 0.0}
        self.calls = {'prefill': 0, 'svd': 0}

    @contextmanager
    def enter(self, kind, stop=None):
        start = time.monotonic()
        acquired = False
        with self.condition:
            if kind == 'prefill':
                self.prefill_waiters += 1
                self.condition.notify_all()
            try:
                while self.active is not None or (kind == 'svd' and self.prefill_waiters):
                    if stop is not None and stop.is_set():
                        break
                    self.condition.wait(timeout=.1)
                else:
                    if stop is None or not stop.is_set():
                        self.active = kind
                        acquired = True
                        self.calls[kind] += 1
                        self.wait_ms[kind] += (time.monotonic() - start) * 1000
            finally:
                if kind == 'prefill':
                    self.prefill_waiters -= 1
                self.condition.notify_all()
        try:
            yield acquired
        finally:
            if acquired:
                with self.condition:
                    self.active = None
                    self.condition.notify_all()

    def metrics(self):
        with self.condition:
            return {**{f'prefill_gate_{k}_wait_ms': v for k, v in self.wait_ms.items()},
                    **{f'prefill_gate_{k}_calls': v for k, v in self.calls.items()}}


def install(mode):
    if mode not in ('sync_control', 'strict'):
        raise ValueError(f'Unknown prefill gate mode: {mode}')
    import torch
    from sglang.srt.managers.scheduler import Scheduler
    from sglang.srt.mem_cache.mamba_radix_cache import MambaRadixCache
    if getattr(Scheduler, '_prefill_gate_mode', None):
        raise RuntimeError('Prefill gate already installed')
    Scheduler._prefill_gate_mode = mode
    gate = PrefillGate()
    original_batch = Scheduler.run_batch
    original_svd = MambaRadixCache._process_compression_batch
    original_info = Scheduler.get_internal_state

    def run_batch(scheduler, batch, *args, **kwargs):
        if not batch.forward_mode.is_extend_or_draft_extend_or_mixed():
            return original_batch(scheduler, batch, *args, **kwargs)
        with gate.enter('prefill') if mode == 'strict' else nullcontext(True):
            try:
                return original_batch(scheduler, batch, *args, **kwargs)
            finally:
                # CPU return does not mean GPU completion. The sync-control arm
                # pays this same boundary cost without excluding SVD.
                event = torch.cuda.Event()
                event.record(torch.cuda.current_stream())
                event.synchronize()

    def svd(tree, batch):
        if mode != 'strict':
            return original_svd(tree, batch)
        with gate.enter('svd', tree._compression_stop_event) as acquired:
            if not acquired:
                return
            # Eviction/reset may cancel a snapshot while we wait. No radix-tree
            # metadata is read by this worker; preserve existing identity checks.
            live = [item for item in batch if tree._is_live_compression_item(item)]
            if live:
                try:
                    return original_svd(tree, live)
                finally:
                    # Original implementation synchronizes on success; also
                    # drain partially submitted work on an exception.
                    if tree._svd_stream is not None:
                        tree._svd_stream.synchronize()

    def info(scheduler, request):
        result = original_info(scheduler, request)
        result.internal_state['cache_observations'].update(prefill_gate_mode=mode, **gate.metrics())
        return result

    Scheduler.run_batch = run_batch
    MambaRadixCache._process_compression_batch = svd
    Scheduler.get_internal_state = info
